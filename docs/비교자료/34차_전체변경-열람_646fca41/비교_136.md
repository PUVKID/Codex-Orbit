+def normalize_metadata(rows):
+    result={}
+    for source in rows:
+        qid=scalar(source,'item').rsplit('/',1)[-1]
+        if not re.fullmatch('Q[1-9][0-9]*',qid): raise ValueError('메타 항목 QID')

+        en,ko=clean(scalar(source,'en')),clean(scalar(source,'ko'))
+        if en: row['title_en']=en
+        if ko or en: row['title']=ko or en

+            value=scalar(source,field).rsplit('/',1)[-1]

+def classify_metadata(result):
+    # 일반 축제 분류와 반복 시리즈의 구조적 근거를 구분합니다.
+    referenced={qid for row in result.values() for qid in row['parent_series']}

+        labels=' '.join(row['type_labels']).casefold()
+        row['kind']=kind_for(labels+' '+row['title_en'])
+        if 'edition' in labels or 'occurrence' in labels or (row['parent_series'] and row['observed_dates']):
+            row.update(item_role='edition',role_evidence='P31 개최분 분류 또는 P179 시리즈 관계와 관측 시점')
+        elif any(text.casefold() in ('recurring event','annual event','biennial event','periodic event') for text in row['type_labels']):
+            row.update(item_role='series',role_evidence='P31에 명시된 반복 행사 분류')
+        elif qid in referenced:
+            row.update(item_role='series',role_evidence='수집한 전체 메타에서 P179 시리즈 참조의 대상')
+        else:row.update(item_role='unclassified',role_evidence='최소 메타에서 반복 시리즈·개최분 구분 근거 부족')
+    return result

+def normalize_paris(source, checked_at, rejections=None):
+    def reject(reason):
+        if rejections is not None:rejections[reason]=rejections.get(reason,0)+1
+        return None
+    if source.get('address_city','').strip().casefold()!='paris':return reject('other_city')
+    title=clean(source.get('title',''))
+    url=source.get('url','')
+    if not title or not isinstance(url,str) or not url.startswith('https://www.paris.fr/evenements/'):return reject('invalid_identity')
+    ident=str(source.get('id',''))
+    if not re.fullmatch('[1-9][0-9]*',ident):return reject('invalid_identity')
+    try:
+        raw_start=datetime.fromisoformat(source['date_start']);raw_end=datetime.fromisoformat(source['date_end'])
+    except (KeyError,TypeError,ValueError):return reject('missing_or_invalid_bounds')
+    if raw_start.tzinfo is None or raw_end.tzinfo is None or raw_end<raw_start:return reject('missing_or_invalid_bounds')
+    occurrences=source.get('occurrences')
+    if not isinstance(occurrences,str) or len(occurrences)>100000:return reject('missing_or_invalid_occurrences')
+    matches=re.findall(r'(\d{4}-\d\d-\d\dT[^_\s;,]+)_(\d{4}-\d\d-\d\dT[^_\s;,]+)',occurrences)
+    if not matches:return reject('missing_or_invalid_occurrences')
+    days=set();conflict=False;date_conflict=False;parsed=[]
+    for start,end in matches:
+        try: a,b=datetime.fromisoformat(start),datetime.fromisoformat(end)
+        except ValueError:return reject('invalid_occurrence_datetime')
+        # 기간 사이를 보간하지 않고 명시된 동일 일자 개최만 수록합니다.
+        if a.date()!=b.date() or not 2026<=a.year<=2099:continue
+        if a.tzinfo is None or b.tzinfo is None or b<a:continue
+        days.add(a.date().isoformat())
+        parsed.append((a,b))
+        if not raw_start.date()<=a.date()<=raw_end.date():date_conflict=True
+    if not days:return reject('no_explicit_single_day_occurrence')
+    for raw,value in [(raw_start,min(parsed,key=lambda pair:pair[0])[0]),(raw_end,max(parsed,key=lambda pair:pair[1])[1])]:
+        if raw.date()==value.date() and (raw.utcoffset()!=value.utcoffset() or raw.time()!=value.time()):conflict=True
+    evidence='official_api_consistent'
+    if conflict or date_conflict:
+        allowed=PARIS_PAGE_DATES.get(ident,set())
+        days=days & allowed
+        if not days:return reject('unverified_date_conflict' if date_conflict else 'unverified_time_conflict')
+        evidence='official_page_crosschecked'
+    return dict(id='paris:'+ident,series_id=None,title=title,title_en=title,kind=kind_for(title+' '+str(source.get('qfap_tags',''))),
+                country_code='FR',city_refs=[dict(wikidata_qid='Q90',geonames_id=2988507)],dates=sorted(days),date_precision='day',
+                status='announced',time_conflict=conflict,date_conflict=date_conflict,source_url=url,license='ODbL-1.0',checked_at=checked_at,
+                source_updated_at=source.get('updated_at'),location_confidence='city_verified',
+                date_evidence=evidence,independent_page_checked_on='2026-11-10' if evidence=='official_page_crosschecked' else None)

++## 33차 · 자동 위치·강수량·도시 축제
++
++위치는 자동 네트워크 추정이 기본입니다. 지역 선택 창에서 자동·수동을 명확히 고릅니다. 수동 검색이나 취소만으로 저장하지 않으며, 자동 모드로 돌아가면 이전 수동 위치를 해제하고 다시 조회합니다. 국가만 고르면 행사 기준만 바뀌고 날씨 지역은 유지됩니다. 네트워크 추정은 GPS나 건물 주소가 아닙니다.
++
++습도 옆에는 강수량(mm)과 제공 자료의 실제 직전 집계 간격을 표시합니다. 실제 0과 미확인(—)을 구분하고, 오래된 값은 이전 정보로 표시합니다. 일출·일몰은 제공 지역의 시간대를 유지합니다.
+ 
+-연도·월·일에 맞춘 3D 테마를 기본으로 보여 주는 Windows 바탕화면 앱입니다. 관측소는 명시적으로 선택해 사용할 수 있습니다.
++화면 위의 선택 영역에서 언어·화면 테마(시스템/라이트/다크)·글자 크기를 바꿀 수 있습니다. 테마는 즉시 적용되고 저장되며, 시스템은 운영체제의 테마 변화를 따릅니다. 왼쪽 작업 현황에서 기록된 작업과 선택 설정·최근 실행에서 확인한 값을 구분해 봅니다. 화면 설정에는 움직임 줄이기·화면 가장자리에 붙이기·알림 소리·성능과 화질·문제 확인용 파일 저장·새 버전 확인이 있습니다. 프라이버시 모드는 화면에서 채팅 제목과 작업 설명을 가리며 저장된 기록은 그대로 유지합니다. 기존 월별 주제와 날짜별 대상은 유지하고, 날짜 위젯의 고리와 좌우 보조 장식은 제거했습니다. 당일 대상 하나를 입체로 표시합니다.
+ 
+-로컬 기록에서 확인한 모델·추론 강도 변화와 작업 상태를 보여 주며, 실제 계정 사용 한도 비율·시계·기온을 함께 확인할 수 있습니다.
++기본 바탕화면은 원래 연도·월·일별 과일·꽃 등의 달력 조형입니다. 모델 변경 알림이 와도 날짜별 제목과 조형을 유지합니다. 입체 관측소는 표시 모드에서 직접 선택할 수 있습니다. [30차 복원 범위](docs/2026-11-02_30차_날짜별테마_기본복원.md)

++## 알림 읽기
+ 
+-날짜 위젯은 고리와 좌우 장식 없이 당일 대상 하나를 표시합니다. 기존 888개 월별 주제와 일반일 대상을 보존하고, 250개 국가·지역의 공휴일·지원 기념일에는 실제 행사명과 특집 조형을 적용합니다. 미래 법 개정과 모든 도시 축제까지 확정한 자료는 아닙니다.
++채팅 제목에는 장식 이모지를 붙이지 않습니다. 표시만 정리하며 Codex의 원래 제목을 바꾸지 않습니다.
+ 
+-지역 표시를 눌러 도시·주소·장소·국가를 직접 검색합니다. 선택 지점의 기온·날씨·습도·일출·일몰을 받아오며, 자동 IP 위치는 추정임을 표시합니다.
++- 채팅 제목
++- ⬅️ 이전 모델 / 이전 추론 강도
++- ➡️ 현재 모델 / 현재 추론 강도
++- 💡 쉬운 이유
+ 
+-알림은 이전·현재 모델과 추론 강도를 나란히 비교하고 짧은 설명과 접을 수 있는 원문으로 정리했습니다. 화면은 흰색·슬레이트 블루와 다크 테마를 제공합니다.
++☀️ Sol · ✨ Astra · 🌍 Terra · 🌙 Luna를 정확한 모델 ID와 함께 표시합니다. 추론 강도는 ⚪ 없음 · 💧 최소 · 🌱 낮음 · ⚖️ 보통 · 🔥 높음 · 🚀 매우 높음 · ⚡ 최고로 구분합니다. 기록되지 않은 이전 값이나 이유는 만들어 쓰지 않습니다. 공개된 배정 설명은 설명으로 표시하며 비공개 사고 과정을 보여 준다고 주장하지 않습니다.

++## 입체 관측소
+ 
+-실제 배포 EXE에서 개인 기록·네트워크 없이 검사한 화면입니다. 기온·습도·잔량과 모델 값은 합성 예시입니다. [검증 범위](docs/공개검증_32차.md)
++- Sol은 태양, Astra는 결정, Terra는 구체, Luna는 초승달 코어입니다. 모델이 미확인이면 중립 코어를 사용합니다.
++- 실제 로컬 기록에서 확인한 작업자는 위성으로 표시합니다. 위성이나 작업 목록을 누르면 해당 채팅 기록을 엽니다. ★ 관심 채팅은 우선 궤도에 표시합니다. 위성은 최대 24개이며 나머지도 목록에서 확인합니다.
++- 진행·완료·응답 대기·오류·중단은 명시된 기록이 있을 때만 표시합니다. 확인되지 않은 작업을 진행 중이나 완료로 추정하지 않습니다.
++- 이전/현재 값은 같은 비중으로 표시하고 실제 바뀐 값만 강조합니다. 선택한 설정과 실행 중 관측한 설정은 구분합니다.
++- 슬라이더는 저장된 기록을 재생합니다. 실제 Codex 모델이나 추론 강도를 되돌리는 기능이 아닙니다.
++- 자리비움 이후 기록 수와 보관한 기록을 확인할 수 있습니다. 채팅 직접 이동은 지원이 확인되지 않아 ID 복사를 제공합니다.

+-## 31차 · 화면 마감과 테마 선택
++잔량은 설치된 Codex의 로그인 상태에서 조회한 **계정 사용 한도 비율**입니다. 정확한 남은 토큰 개수가 아닙니다. 숫자와 막대, 관측소의 입체 용기로 표시합니다. 미확인은 0%와 다르며, 오래된 값은 조회 상태와 함께 구분합니다.
+ 
+-시스템·라이트·다크 화면 테마를 즉시 적용하고 선택을 저장합니다. 언어·글자 크기·회전 속도는 같은 설정 영역에서 조절하고, 작업 현황에서 선택 설정과 최근 실행 관측을 구분합니다. 화면 설정 메뉴는 움직임 줄이기·가장자리 붙이기·알림 소리·성능과 화질·문제 확인용 파일 저장·새 버전 확인을 설명합니다. 프라이버시 모드는 화면의 제목과 작업 설명을 가리며 저장 원문을 유지합니다. 날짜별 과일·세 금속 링·날짜 규칙과 모델·추론 변경 알림은 유지합니다.
++위젯의 지역 표시를 눌러 도시·지역, 주소·장소 또는 국가를 검색하고 결과에서 직접 선택합니다. 도시·장소를 고르면 해당 지점의 현재 기온·날씨·습도·강수량·일출·일몰을 네트워크로 받습니다. 국가만 고르면 디자인 국가만 바꾸고 날씨 기준 지역은 별도로 표시합니다. 자동 위치는 IP 기반 추정이므로 VPN·통신망에 따라 다른 도시가 나올 수 있습니다. Windows 위치 서비스를 켤 필요가 없습니다. 상세 주소는 검색 결과에 있는 정보만 표시하며 실제 건물 위치를 자동으로 보장하지 않습니다. 날씨는 선택 좌표 주변의 기상 자료이며 건물마다 별도의 실측 기온을 제공하는 것은 아닙니다. 연결 실패와 오래된 값은 갱신 상태로 구분합니다.

+-## 감시·표시·저장·업데이트의 연결
++## MD·진단·업데이트
+ 
+-30차 소스에서 `notifier.Watcher`는 로컬 JSONL을 읽어 설정 관측·실행 관측·공개 이유를 구분하고 `app.py`가 알림·위젯·상세 이력을 표시합니다. `core.py`는 JSONL 꼬리 읽기·채팅 제목·화면 위치와 Markdown 적용/복구를 담당합니다. `activity.py`의 작업 상태와 `notice_format.py`의 표현을 렌더러의 장식 상태와 구분합니다.
++배포 폴더의 `기본_지침.md`는 공유 가능한 시작 자료입니다. 개인 MD를 다른 사람에게 줄 때 대화·연락처·인증값을 먼저 제외하십시오. 위젯이 Codex의 모든 채팅에 MD를 강제로 주입하거나 비공개 추론을 읽는 것은 아닙니다. Codex에 적용하려면 해당 프로젝트가 지원하는 지침 파일로 사용해야 합니다.

+-| 배포 교체 | safe_update의 check→stage→apply 단계로 공식 릴리스·자산·SHA·로컬 경계를 검사하고 실행 확인 실패 시 이전 EXE를 복구합니다. MAX_DOWNLOAD는512MiB입니다. |
++‘문제 확인용 파일 저장’은 대화·이벤트 원문·인증값·개인 경로를 제외한 허용 항목만 **로컬 파일**로 저장합니다. 자동 전송하지 않습니다.
+ 
+-사용자는 감시 경로·언어·표시 모드·모션/자원 품질을 고르고 새 알림의 이전/이후 값과 근거를 읽습니다.
++‘새 버전 확인’은 공식 `PUVKID/Codex-Orbit` HTTPS 릴리스만 확인합니다. 새 버전을 내려받아 SHA-256을 검증한 뒤 직접 설치·재시작할 수 있습니다. 재시작 검사 실패 시 이전 실행 파일을 복구하며 개인 설정과 MD는 보존합니다. 소스 실행에서는 배포 EXE의 설치 동작을 대신 수행하지 않습니다.
+ 
+-날씨·사용 한도·업데이트는 네트워크 또는 App Server 기능이며 로컬 감시기의 기능 범위와 나눠 봅니다. 방송 모드와 진단 ZIP의 공개 제외 조건을 유지합니다. 실제 사용법·릴리스·빌드 안내와 차수별 함수/설정 참조는 아래에 연결되어 있습니다.

+-## 30차 · 원래 날짜별 테마 기본 복원
++필요 환경: Windows 10/11 x64, OpenGL 3.3 지원 그래픽 드라이버. 개인 데이터는 사용자 AppData의 CodexOrbit 폴더에 유지하며 제거 프로그램은 개인 기록을 임의로 지우지 않습니다. 출처·라이선스는 배포의 라이선스 폴더를 확인하십시오.
+ 
+-설정에 표시 모드가 없는 경우에도 원래 날짜별 달력 조형을 표시합니다. 모델 변경 이벤트를 받아도 날짜 제목과 조형은 유지합니다. 888개 월 테마·일별 디자인·색·재질·날짜 규칙은 바꾸지 않았습니다. 이전/현재 모델·추론 강도·알림 기능도 유지합니다.

+-## 선택 가능한 29차 관측소 기능
+-
+-- 은색 금속 링 세 개가 서로 다른 축에서 움직입니다. 솔의 태양·아스트라의 결정·테라의 구체·루나의 초승달 코어와 추론 강도별 입자 밀도를 구분합니다. 확인되지 않은 모델은 중립 조형으로 표시합니다.
+-- 위성은 기록에서 명시적으로 관측한 작업 상태를 표시합니다. 관심 채팅 고정·기록 북마크·자리비움 요약·시간 슬라이더 재생을 제공하며 재생은 실제 모델 설정을 바꾸지 않습니다.
+-- 알림은 채팅 제목·이전 모델과 추론 강도·현재 모델과 추론 강도·기록된 이유를 비교합니다. 정확한 모델 ID와 원문 기록은 보존하고 표시 제목의 장식 이모지를 정리합니다. 마우스를 올리면 자동 닫힘을 멈추고 알림을 고정할 수 있습니다.
+-- 방송 모드는 열린 알림·제목·작업·툴팁을 가립니다. 진단 ZIP은 대화·인증·경로·이벤트 원문을 제외한 항목만 로컬에 저장합니다. 업데이트는 사용자가 확인을 눌러 공식 릴리스를 조회하고 SHA-256 대조 후 준비하며, 교체 후 실행 확인 실패 시 이전 실행 파일로 복구합니다.
+-- 민트와 화이트 화면·나눔고딕 Bold를 사용합니다. 글자 크기·가장자리 도킹·펼침·재생/일시정지·모션 축소·자원 품질을 조절할 수 있으며 일시정지 중에도 알림 감시는 계속됩니다.
+-
+-실제 한도는 **남은 사용 한도 비율**이며 남은 토큰 개수가 아닙니다. 미확인은 0%와 구분합니다. 선택 설정 관측과 실제 실행 기록 관측을 구분하며 로컬 기록이 서버 모델 신원을 보증하지 않습니다. 기록되지 않은 이유와 작업 상태를 만들어 내지 않습니다.
+-
