+import hashlib
+import json
+import os
+from pathlib import Path
+import shutil
+import subprocess
+import sys
+import threading
+import time
+import zipfile
+
+ROOT = Path(__file__).resolve().parents[1]
+
+
+def run(argv, name, timeout):
+    """실제 인자와 종료 결과를 별도 파일에 보존합니다."""
+    started = time.time()
+    build_env = os.environ.copy()
+    base_python = Path(sys.base_prefix)
+    windows = Path(os.environ.get('SystemRoot', r'C:\Windows'))
+    # 외부 PDF 도구 등의 동명 ICU DLL이 Qt 의존성으로 들어오지 않게 합니다.
+    build_env['PATH'] = os.pathsep.join(map(str, [Path(sys.executable).parent, base_python,
+                                                windows / 'System32', windows]))
+    process = subprocess.Popen(argv, cwd=ROOT, stdout=subprocess.PIPE,
+                               stderr=subprocess.STDOUT, shell=False, env=build_env)
+    captured = bytearray()
+    total_bytes = 0
+    def collect():
+        nonlocal total_bytes
+        while True:
+            chunk = process.stdout.read(65536)
+            if not chunk:
+                break
+            total_bytes += len(chunk)
+            captured.extend(chunk[:max(0, 524288 - len(captured))])
+    reader = threading.Thread(target=collect, daemon=True)
+    reader.start()
+    timed_out = False
+    try:
+        exit_code = process.wait(timeout=timeout)
+    except subprocess.TimeoutExpired:
+        timed_out = True
+        subprocess.run(['taskkill', '/PID', str(process.pid), '/T', '/F'],
+                       stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=15)
+        exit_code = process.wait(timeout=15)
+    reader.join(timeout=5)
+    process.stdout.close()
+    raw = bytes(captured)
+    (ROOT / 'work' / (name + '.raw')).write_bytes(raw)
+    try:
+        decoded = raw.decode('utf-8')
+        valid_utf8 = True
+    except UnicodeDecodeError:
+        decoded = raw.decode('utf-8', errors='replace')
+        valid_utf8 = False
+    output = decoded.encode('utf-8')
+    (ROOT / 'work' / (name + '.log')).write_bytes(output)
+    record = {'argv': argv, 'cwd': str(ROOT), 'exit_code': exit_code, 'timed_out': timed_out,
+              'started_at': started, 'elapsed_seconds': time.time() - started,
+              'output_bytes': total_bytes, 'captured_bytes': len(raw),
+              'valid_utf8': valid_utf8, 'output_truncated': total_bytes > len(raw)}
+    (ROOT / 'work' / (name + '.json')).write_text(
+        json.dumps(record, ensure_ascii=False, indent=2), encoding='utf-8')
+    print(json.dumps(record, ensure_ascii=False), flush=True)
+    if exit_code or timed_out:
+        print(output[-7000:].decode('utf-8', errors='replace'))
+        raise SystemExit(exit_code or 1)
+

+## 디자인 기준과 범위
+

+# Codex Orbit 사용 설명 · 10차 / 버전 1.5.2
+
+실행하면 투명 배경의 무채색 입체 궤도가 화면 위에 나타납니다. 세 원형 궤도는 XYZ 회전·원근 투영·깊이순 가림과 금속 명암으로 움직입니다. 외곽과 중앙 허브는 1:1 원형입니다. 나눔고딕 Bold 글꼴을 내장하며 인터넷으로 대화 내용을 전송하지 않습니다.
+
+## 궤도 조작
+
+- 마우스로 드래그하면 위치를 옮깁니다. 다음 실행에서도 위치를 기억하고, 모니터가 바뀌어 화면 밖에 놓이면 보이는 화면으로 복구합니다.
+- 궤도를 클릭하면 변경 이력이 열립니다. 우클릭하면 설정과 종료 메뉴가 나옵니다.
+- 중심 아래 작은 점은 정상 감시일 때 회색, 새 변경이 있을 때 흰색이며, 감시 오류·준비 중에는 어두운 점 안에 가로선을 표시합니다. 정확한 감시 상태 문구는 상세 창에서 확인합니다.
+- 새 변경의 말풍선을 클릭하면 그 알림의 정확한 이력을 엽니다. 새 항목이 추가돼도 다른 항목으로 바뀌지 않습니다. 기본 표시 시간은 30초이며, 설정의 ‘알림을 직접 닫을 때까지 유지’를 켜면 닫기를 누를 때까지 유지합니다.
+- 변경마다 별도 말풍선이 즉시 나타납니다. 앞 알림을 닫을 필요가 없습니다. 화면 안에 세로로 쌓고 공간이 부족하면 어긋나게 겹칩니다. 새 알림을 가장 위에 표시하며 각 창의 닫기·클릭·자동 만료는 다른 창에 영향을 주지 않습니다. 같은 ID의 갱신은 기존 창을 보강하고 표시 시간을 초기화하지 않습니다. 이력 다시 보기는 같은 창이 있으면 갱신하고 없으면 즉시 새 창으로 엽니다.
+- 동시에 최대 8장을 표시합니다. 초과하면 가장 오래된 창만 숨기고 새 알림은 바로 표시합니다. 숨긴 누적 건수는 말풍선과 이력 상단에 안내합니다. 이력에서 다시 볼 수 있으며 화면 이력은 최근 500건, 파일 보존에는 아래 순환 상한이 적용됩니다.
+- 채팅 제목을 맨 위에 크게 표시하고 모델 변경·추론 강도 변경·모델·추론 변경을 구별합니다. 이전 값과 새 설정을 위아래로 비교하며 새 설정을 강조합니다. 강도는 없음 (none), 최소 (minimal), 낮음 (low), 보통 (medium), 높음 (high), 매우 높음 (xhigh), 최고 (ultra)처럼 원문과 함께 표시합니다. 한글은 앱의 표시 편의이며 공식 품질 보증이 아닙니다. 모델 ID·공개 이유 원문은 상세에 보존하고 확인되지 않은 이유는 만들어 넣지 않습니다.
+- 같은 감시 묶음은 유효한 원본 시각 순으로 전달하고 시각이 같으면 감지 순서를 유지합니다. 시각이 없거나 잘못되면 감지 위치를 유지하고 이미 표시한 알림은 뒤늦게 읽은 파일로 재정렬하지 않습니다.
+- 밝은 화면에서는 ‘밝은 배경’, 어두운 화면에서는 ‘어두운 배경’ 대비를 선택할 수 있습니다.

+## 실시간 Codex 잔량
+
+외곽 눈금·진행 링·빛 이동은 장식 모션입니다. 중간 애니메이션 값을 계정 수치로 표시하지 않습니다. 확인된 숫자는 반올림 애니메이션 없이 즉시 표시합니다.
+
+중심 숫자는 현재 로그인된 Codex 계정의 남은 **사용 한도 비율**입니다. 남은 토큰 개수를 뜻하지 않습니다. 여러 버킷·기간이 있으면 확인된 값 중 가장 적게 남은 비율을 중심에 표시합니다. 상세 창 상단에서 각 버킷의 기간별 잔량 막대·초기화 한국 시각·최종 조회 시각을 확인할 수 있습니다. 서버가 주간 한도만 주면 주간만 표시하며 없는 5시간 한도를 만들지 않습니다.
+
+앱은 PC에 설치된 Codex의 공식 App Server를 별도 숨김 프로세스로 연결하여 읽기 전용 `account/rateLimits/read`만 조회합니다. 최초 실행 후 30초마다 자동 조회하며 ‘새로고침’을 눌러 수동 조회할 수 있습니다. 서버가 사용 한도 변경을 알리면 반영합니다. 실제 조회 지연과 서버 반영 시차가 있으며 매 순간의 확정값은 아닙니다. 연결 중에도 궤도 회전·드래그와 기존 알림 기능을 사용할 수 있습니다.
+
+아직 한도를 받지 못했거나 수치가 없으면 중심은 ‘—’입니다. 조회 실패·지연 또는 초기화 시각 경과 후 재조회 전에는 과거 수치를 현재값으로 표시하지 않으며, 이전 수치가 있으면 중심에 ‘?’와 상세에 ‘이전 조회 값’을 표시합니다. 초기화 시각이 지났다고 임의로 100%로 바꾸지 않습니다. 잔량을 디스크에 캐시해 다음 실행에 복원하지 않습니다.
+
+설치된 Codex에서 ChatGPT 로그인이 되어 있어야 합니다. 앱은 로그인·계정 선택·리셋·크레딧 구매·모델 실행·Codex 설정 변경을 하지 않으며 인증 파일을 직접 읽거나 복사하지 않습니다. 연결 실패 시 세 차례까지만 자동 재연결하며 수동 새로고침으로 다시 시도할 수 있습니다. Codex EXE 자동 탐색이 실패하면 ‘설정 · 진단’의 ‘Codex EXE 선택’에서 설치된 `codex.exe`를 선택하고 저장하세요. 비워 저장하면 자동 탐색으로 돌아갑니다. API 키를 입력하는 기능은 없습니다.

+## 모델 변경 감시와 근거
+
+기본은 CODEX_HOME 환경변수 또는 사용자 홈의 .codex/sessions 폴더입니다. 설정에서 실제 sessions 폴더를 선택할 수 있습니다. 최초 실행은 기존 기록을 기준선으로 읽으므로 과거 알림을 쏟아내지 않습니다. 폴더별 상태가 별도로 보존됩니다.
+
+Codex가 채팅 ID와 함께 남긴 선택 설정 기록을 읽어 모델·추론 강도 변경을 다음 메시지 실행 전에 감지합니다. ‘채팅 선택 설정 변경’은 설정 관측이며 사용자가 바꿨다는 사실이나 서버 실행 모델을 인증하지 않습니다. 일반·계획 모드의 활성 모델과 추론 강도를 비교하고, 같은 설정 반복이나 뒤따르는 동일 turn_context는 중복 알림을 만들지 않습니다. 실제 로컬 실행 관측이 선택값과 다르면 별도로 남깁니다. 부모에서 상속한 다른 채팅의 설정·턴은 자식의 변경으로 처리하지 않습니다.
+
+감시 주기 대기는 200ms입니다. Codex가 기록하기 전의 화면 선택이나 서버 내부 변경을 알 수 없으며, 큰 파일을 나누어 읽는 시간과 화면 처리 시간 때문에 전체 알림 지연이 200ms 이하라고 보장하지 않습니다. 같은 sessions 루트 아래 날짜·프로젝트·자식 채팅을 감시하지만 탐색은 기존처럼 최대 4096개 파일로 제한됩니다.
+
+이전 감시 상태를 처음 열면 원본 state를 별도 백업하고 이미 읽은 위치까지만 선택 설정 기준선을 재생합니다. 과거 이력·중복 방지 기록을 초기화하지 않습니다. 재생 시작 뒤 추가된 설정은 새 변경으로 처리하며 중간 종료 후에는 저장된 위치에서 이어갑니다. 남은 기존 기록이 있으면 ‘기준선 준비 중’과 잔여 바이트를 진단에 표시합니다.
+
+감지 시각은 앱이 변경을 발견한 시각이며, 원본 turn_context의 시각은 별도 보존해 상세에서 표시합니다. 파일 읽기 예산 때문에 뒤늦게 발견한 항목과 이미 전송한 알림 사이의 전체 시간순까지 보장하지 않습니다. 비교할 수 없는 시각을 현재 시각으로 만들어 정렬하지 않습니다.
+
+채팅 제목은 sessions와 같은 상위 폴더의 session_index.jsonl에서 읽습니다. 앱 데이터 폴더의 title-cache.json에 공식 앱에서 확인한 제목을 저장하면 그 확인 캐시를 우선합니다. 출처와 확인 시각은 상세에 남으며 현재 실시간 조회로 표현하지 않습니다. 자식 작업에는 부모 채팅의 정확한 제목과 별도 작업 이름을 표시합니다. 확인할 수 없으면 ‘채팅 제목 확인 불가’로 표시하며 첫 프롬프트나 폴더명을 제목으로 만들지 않습니다.
+
+공개 이유는 부모의 공개 설명과 spawn 호출 ID·호출 결과·자식 작업 경로가 정확히 연결된 첫 배정에서만 보강합니다. 명시 요청에 모델·강도가 있으면 관측값과 대조합니다. 부모가 늦게 읽히면 기존 이력을 보강하며 과거 알림을 다시 만들지 않습니다. 최근 부모 기록은 파일당 최대 4 MiB, 주기당 추가 8 MiB 안에서 우선 조회합니다. 범위 밖의 호출은 순차 감시를 기다리므로 즉시 이유를 찾지 못할 수 있습니다. 기록에 공개 이유가 없거나 연결 근거가 부족하면 ‘확인 불가’입니다. 다른 도구가 공개 설명과 배정 사이에 있으면 자동으로 이유를 추정하지 않습니다. 로컬 기록이나 명시 배정 요청은 서버 모델 신원 인증이 아닙니다. 비공개 추론을 추출하지 않습니다.

+- `CodexOrbit.exe`: Python 설치 없이 실행하는 단일 파일입니다. 기본 지침, 사용법, 글꼴이 내장되어 있습니다.
+- `CodexOrbit-Setup.exe`: 현재 사용자용 설치본입니다. 설치 위치를 선택하고 시작 메뉴 바로가기와 제거 기능을 사용할 수 있습니다.
+- `기본_지침.md`: 개인 정보가 빠진 배포 기본 지침입니다. 자신의 MD를 대신 가져와도 됩니다.
+- `사용법.md`, `배포안내.md`: 한국어 사용 문서입니다.
+- `라이선스/`, `소스와_재빌드.zip`: 포함 구성 요소의 라이선스와 소스·재빌드 자료입니다. 재배포할 때도 함께 전달하십시오.

+## 시작과 MD 적용
+
+앱을 켜면 작은 회전 위젯이 나타납니다. 끌어서 옮기고 클릭해서 상세 창을 여십시오. 처음에는 기존 기록을 기준선으로 읽으며 과거 알림을 쏟아내지 않습니다. 이전 버전의 감시 상태는 원본을 백업하고 이미 읽은 위치까지만 선택 설정 기준선을 복원합니다. 기존 이력을 지우지 않으며 복원 중 추가된 변경은 이후 새 알림으로 처리합니다. 큰 기록은 ‘기준선 준비 중’으로 표시하며 처리 시간이 걸릴 수 있습니다.
+
+MD 탭에서 지침을 가져오거나 편집하고 적용 대상 AGENTS.md를 선택합니다. 미리보기를 확인하고 적용 버튼을 누릅니다. 기존 내용은 백업하고 관리구역만 갱신합니다. 다른 프로그램이 파일을 변경하면 적용·원복을 거부하여 그 변경을 보호합니다. 다른 사람에게 윤석님의 원본 MD를 줄 때는 개인 이름·절대 경로·프로젝트·민감 정보부터 확인하십시오.
+
+MD 파일 적용은 Codex의 설정·권한·인증을 바꾸거나 이미 열린 모든 채팅에 지침을 재주입하는 기능이 아닙니다. 새 채팅이나 해당 프로젝트를 다시 여는 시점의 지침 읽기를 통해 적용 여부를 확인하십시오. 앱은 모델 자체를 변경하지 않습니다.
+
+## 기록과 한계
+
+감시 대상은 현재 사용자의 CODEX_HOME 또는 기본 .codex/sessions입니다. 설정에서 경로를 바꿀 수 있습니다. 로컬 기록이 생성되어야 변경을 관측할 수 있으므로 설정 화면을 누르는 순간과 감지 시각은 다를 수 있습니다. 변경 이유가 명시적으로 연결되지 않으면 확인 불가로 표시합니다. 서버 실행 모델 인증과 비공개 내부 사고과정은 제공하지 않습니다. 표시되는 추론 정보는 기록된 추론 강도와 공개 변경 이유입니다. 네트워크로 대화나 MD를 보내지 않습니다.
+
+‘채팅 선택 설정 변경’은 Codex가 해당 채팅 ID와 함께 남긴 설정 기록을 뜻합니다. 다음 실행의 로컬 모델 관측과 구분하며 변경 주체나 이유를 추정하지 않습니다. 같은 설정의 반복 기록과 뒤따르는 동일 실행은 중복 알림을 만들지 않습니다. 실행 관측이 선택값과 다르면 별도 관측으로 남깁니다. 감시 주기 대기는 200ms이며, 초기 기준선·대용량 파일·파일 읽기 예산과 화면 처리 시간 때문에 전체 지연이 200ms 이내라고 보장하지 않습니다.
+
+사용자별 앱 데이터와 이력은 사용자 데이터 폴더에 보존됩니다. 설치본 제거는 프로그램 파일과 바로가기를 제거하며 앱이 적용한 지침·백업·이력을 자동 삭제하지 않습니다.

+## 라이선스와 재빌드
+
+Python, PySide6/Qt, PyInstaller와 나눔고딕을 사용합니다. 나눔고딕은 SIL OFL, PySide6/사용 Qt 라이브러리는 LGPLv3 등 각 구성 요소 라이선스를 따릅니다. Qt 수정본 분석과 교체를 막는 제한을 추가하지 않습니다. 함께 제공한 소스에서 `python -m venv work/venv`, 가상환경의 `python -m pip install -r requirements.txt`로 환경을 준비하고 `python -m PyInstaller --onefile --windowed --name CodexOrbit --add-data "assets;assets" --paths src src/app.py`로 재빌드할 수 있습니다. 다른 호환 PySide6/Qt를 설치한 뒤 같은 소스로 재빌드해 교체할 수 있습니다. 설치본 재빌드는 Inno Setup 6.7.3의 ISCC로 `tools/installer.iss`를 컴파일합니다. 자세한 구성과 고정 버전은 소스 압축 안 requirements.txt와 tools/build_distribution.py에 있습니다.
+
+프로그램은 코드 서명 인증서 없이 제작한 배포본입니다. 사용자 PC의 보안 검사 결과나 실행 허용을 보장하지 않으며 보안 기능을 끄지 않습니다.
