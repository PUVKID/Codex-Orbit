+FRAME_COUNT = 540
+FRAME_SIZE = 296
+FPS = 30
+PERIOD_SECONDS = 18
+CACHE_LIMIT = 24
+MAX_ARCHIVE_BYTES = 128 * 1024 * 1024
+MAX_FRAME_BYTES = 1024 * 1024
+MAX_TOTAL_BYTES = 256 * 1024 * 1024

+class AtomAssetError(RuntimeError):
+    """필수 시각 자산 오류를 시작 실패 경로에 전달합니다."""

+def asset_directory():
+    root = Path(getattr(sys, '_MEIPASS', Path(__file__).resolve().parents[1]))

+def _identity(path):
+    value = path.stat()
+    return value.st_dev, value.st_ino, value.st_size, value.st_mtime_ns

+def _json_object(pairs):
+    value = {}
+    for key, item in pairs:
+        if key in value:
+            raise AtomAssetError('원자 자산 매니페스트에 중복 필드가 있습니다.')
+        value[key] = item
+    return value

+class AtomFrames:
+    """시작 시 전체 무결성을 검사하고 해제된 QImage는 최대24장만 보존합니다."""
+    def __init__(self, directory):
+        # 제품 호출은 asset_directory만 사용하며 별도 경로는 고립 fixture용입니다.
+        self.directory = Path(directory)
+        self.cache = OrderedDict()
+        self.archive = None
+        self._stream = None
+        try:
+            self._prepare()
+        except Exception as error:
+            self.close()
+            if isinstance(error, AtomAssetError):
+                raise
+            raise AtomAssetError('필수 금속 원자 자산을 읽을 수 없습니다: ' + str(error)) from error

+    def _prepare(self):
+        manifest_path = self.directory / 'manifest.json'
+        archive_path = self.directory / 'frames.zip'
+        if not 0 < manifest_path.stat().st_size <= MAX_MANIFEST_BYTES:
+            raise AtomAssetError('원자 자산 매니페스트 크기가 허용 범위를 벗어났습니다.')
+        self.manifest = json.loads(manifest_path.read_text(encoding='utf-8'), object_pairs_hook=_json_object)
+        expected = {'schema_version': 1, 'width': FRAME_SIZE, 'height': FRAME_SIZE,
+                    'fps': FPS, 'period_seconds': PERIOD_SECONDS, 'frame_count': FRAME_COUNT,
+                    'frame_pattern': 'frame_{:04d}.png', 'archive': 'frames.zip'}
+        if not isinstance(self.manifest, dict):
+            raise AtomAssetError('원자 자산 매니페스트는 객체여야 합니다.')
+        for key, value in expected.items():
+            actual = self.manifest.get(key)
+            if type(actual) is not type(value) or actual != value:
+                raise AtomAssetError('원자 자산 계약 불일치: ' + key)
+        for key in ('geometry', 'lighting', 'source'):
+            if not isinstance(self.manifest.get(key), (dict, str)) or not self.manifest[key]:
+                raise AtomAssetError('원자 자산 제작 근거 누락: ' + key)
+        digest = self.manifest.get('archive_sha256')
+        if not isinstance(digest, str) or re.fullmatch('[0-9a-f]{64}', digest) is None:
+            raise AtomAssetError('원자 프레임 ZIP의 SHA256 형식이 잘못됐습니다.')
+        if not 0 < archive_path.stat().st_size <= MAX_ARCHIVE_BYTES:

+        self.identities = (_identity(manifest_path), _identity(archive_path))
+        self._stream = archive_path.open('rb')
+        hasher = hashlib.sha256()
+        while chunk := self._stream.read(1024 * 1024):
+            hasher.update(chunk)
+        if hasher.hexdigest() != digest:
+            raise AtomAssetError('원자 프레임 ZIP의 SHA256이 일치하지 않습니다.')
+        self._stream.seek(0)
+        self.archive = zipfile.ZipFile(self._stream)

+        names = {f'frame_{index:04d}.png' for index in range(FRAME_COUNT)}

+            raise AtomAssetError('원자 프레임 ZIP의 정확한540개 이름·목록이 일치하지 않습니다.')
+        if sum(item.file_size for item in members) > MAX_TOTAL_BYTES:
+            raise AtomAssetError('원자 프레임 전체 읽기 상한을 초과했습니다.')
+        for item in members:

+                    not 0 < item.file_size <= MAX_FRAME_BYTES):
+                raise AtomAssetError('원자 프레임 멤버의 형식·크기가 허용 범위를 벗어났습니다.')
+            # read는 각 멤버의 CRC도 검사합니다. 이미지는 한 장씩 검사하고 버립니다.
+            image = self._decode(item.filename)
+            if item.filename == 'frame_0000.png':

+    def _decode(self, name):
+        with self.archive.open(name) as stream:
+            data = stream.read(MAX_FRAME_BYTES + 1)
+        if len(data) > MAX_FRAME_BYTES or len(data) < 33 or data[:8] != b'\x89PNG\r\n\x1a\n':
+            raise AtomAssetError('원자 프레임 PNG 서명·크기가 잘못됐습니다: ' + name)
+        if (data[12:16] != b'IHDR' or struct.unpack('>II', data[16:24]) != (FRAME_SIZE, FRAME_SIZE)
+                or data[24:26] != bytes((8, 6))):
+            raise AtomAssetError('원자 프레임은296x296의8비트RGBA PNG여야 합니다: ' + name)
+        image = QImage.fromData(data, 'PNG')
+        if (image.isNull() or image.width() != FRAME_SIZE or image.height() != FRAME_SIZE
+                or not image.hasAlphaChannel() or not image.allGray()):
+            raise AtomAssetError('원자 프레임의 디코딩·무채색·알파 계약이 잘못됐습니다: ' + name)
+        if any(image.pixelColor(x, y).alpha() for x, y in ((0, 0), (295, 0), (0, 295), (295, 295))):
+            raise AtomAssetError('원자 프레임의 모서리는 투명해야 합니다: ' + name)
+        return image.convertToFormat(QImage.Format.Format_ARGB32_Premultiplied)
+
+    def _check_unchanged(self):
+        actual = (_identity(self.directory / 'manifest.json'), _identity(self.directory / 'frames.zip'))
+        if actual != self.identities:
+            raise AtomAssetError('검증 뒤 원자 자산이 변경됐습니다. 앱을 다시 실행해 주세요.')
+
+    @staticmethod
+    def frame_index(elapsed):
+        if isinstance(elapsed, bool) or not isinstance(elapsed, (int, float)) or not math.isfinite(elapsed):
+            raise AtomAssetError('원자 프레임의 재생 시각이 유한한 숫자가 아닙니다.')
+        return int((elapsed % PERIOD_SECONDS) * FPS) % FRAME_COUNT
+
+    def frame(self, elapsed):
+        self._check_unchanged()
+        index = self.frame_index(elapsed)
+        if index not in self.cache:
+            try:
+                self.cache[index] = self._decode(f'frame_{index:04d}.png')
+            except Exception as error:
+                raise AtomAssetError('원자 프레임 재생 실패: ' + str(error)) from error
+            while len(self.cache) > CACHE_LIMIT:
+                self.cache.popitem(last=False)
+        self.cache.move_to_end(index)
+        return self.cache[index]
+
+    @property
+    def cache_bytes(self):
+        return sum(image.sizeInBytes() for image in self.cache.values())
+
+    def close(self):
+        if self.archive is not None:
+            self.archive.close()
+            self.archive = None
+        if self._stream is not None:
+            self._stream.close()
+            self._stream = None
+        self.cache.clear()

+class BrandClock(QObject):
+    """보이는 로고만 단일 시계에서 같은 프레임으로 갱신합니다."""
+    def __init__(self):
+        super().__init__(QApplication.instance())
+        self.members = weakref.WeakSet()
+        self.origin = time.monotonic()
+        self.timer = QTimer(self)
+        self.timer.setInterval(1000)
+        self.timer.timeout.connect(self.prune)
+        self.last_serial=None
+
+    @property
+    def active_count(self):
+        return sum(isValid(mark) and mark.isVisible() for mark in self.members)
+
+    def watch(self, mark):
+        self.members.add(mark)
+        from calendar_motion import set_calendar_motion_visible,set_calendar_motion_size
+        set_calendar_motion_visible(mark, True)
+        set_calendar_motion_size(mark,max(32,min(2160,round(mark.width()*mark.devicePixelRatioF()))))
+        if not self.timer.isActive():
+            self.timer.start()
+        self.tick()
+
+    def unwatch(self, mark):
+        self.members.discard(mark)
+        from calendar_motion import set_calendar_motion_visible
+        set_calendar_motion_visible(mark, False)
+        self.prune()
+
+    def prune(self, *_):
+        if not self.active_count:
+            self.timer.stop()
+
+    def tick(self):
+        marks = [mark for mark in self.members if isValid(mark) and mark.isVisible()]
+        if not marks:
+            self.timer.stop()
+            return
+        elapsed = time.monotonic() - self.origin
+        from calendar_motion import get_calendar_frame,record_calendar_presentation
+        source = get_calendar_frame(elapsed)
+        serial=source.text('calendar_frame_id')
+        if serial and serial==self.last_serial:return
+        self.last_serial=serial
+        pixmaps = {}
+        for mark in marks:
+            size = mark.width()
+            if size not in pixmaps:
+                image = motion_brand_source(source,size)
+                pixmaps[size] = QPixmap.fromImage(image.scaled(size, size, Qt.AspectRatioMode.KeepAspectRatio,
+                                                              Qt.TransformationMode.SmoothTransformation))
+            mark.setPixmap(pixmaps[size])
+            record_calendar_presentation(mark,source)

+@lru_cache(maxsize=32)
+def symbol_image(kind, size=20):

+    source = validate_symbols()[name]
+    points = [(x,y) for y in range(96) for x in range(96) if source.pixelColor(x,y).alpha() > 0]
+    if not points:
+        raise AtomAssetError('3D 기능 기호에 보이는 형상이 없습니다: '+name)
+    left, right = min(x for x,_ in points), max(x for x,_ in points)
+    top, bottom = min(y for _,y in points), max(y for _,y in points)
+    bounds = QRect(max(0,left-2),max(0,top-2),min(95,right+2)-max(0,left-2)+1,
+                   min(95,bottom+2)-max(0,top-2)+1)
+    cropped = source.copy(bounds).scaled(size,size,Qt.AspectRatioMode.KeepAspectRatio,
+                                        Qt.TransformationMode.SmoothTransformation)

+    image = QImage(size,size,QImage.Format.Format_ARGB32_Premultiplied)
+    image.fill(Qt.GlobalColor.transparent)
+    painter = QPainter(image)
+    painter.drawImage((size-cropped.width())//2,(size-cropped.height())//2,cropped)
+    painter.end()
+    return image

+def rotation(axis, angle):
+    """정규화된 실제 3D축을 중심으로 로드리게스 회전행렬을 만듭니다."""
+    axis = np.asarray(axis, dtype=np.float64)
+    axis /= np.linalg.norm(axis)
+    x, y, z = axis
+    c, s, t = (math.cos(angle), math.sin(angle), 1 - math.cos(angle))
+    return np.array([[t * x * x + c, t * x * y - s * z, t * x * z + s * y], [t * x * y + s * z, t * y * y + c, t * y * z - s * x], [t * x * z - s * y, t * y * z + s * x, t * z * z + c]], dtype=np.float64)

+def sphere_mesh(radius, longitude=96, latitude=48):
+    vertices = []
+    indices = []
+    for i in range(latitude + 1):
+        phi = math.pi * i / latitude
+        for j in range(longitude + 1):
+            theta = math.tau * j / longitude
+            n = (math.sin(phi) * math.cos(theta), math.cos(phi), math.sin(phi) * math.sin(theta))
+            vertices.append((*[radius * v for v in n], *n))
+    for i in range(latitude):
+        for j in range(longitude):
+            a = i * (longitude + 1) + j
+            b = a + longitude + 1
+            indices.extend((a, a + 1, b, a + 1, b + 1, b))
+    return (np.asarray(vertices, dtype='f4'), np.asarray(indices, dtype='u4'))

+def finish(vertices, indices):
+    """표면 법선과 삼각형 winding을 일치시킵니다."""
+    v = np.asarray(vertices, dtype='f4')
+    ix = np.asarray(indices, dtype='u4').reshape(-1, 3)
+    p = v[ix, :3]
+    n = v[ix, 3:].mean(axis=1)
+    reverse = np.einsum('ij,ij->i', np.cross(p[:, 1] - p[:, 0], p[:, 2] - p[:, 0]), n) < 0
+    ix[reverse] = ix[reverse][:, [0, 2, 1]]
+    return (v, ix.reshape(-1))

+def box(size, bevel=3, steps=14):
+    """육면체 면을 실제 둥근 모서리와 구면 꼭짓점으로 투영합니다."""
+    half = np.asarray(size, dtype=float) / 2
+    inner = half - bevel
+    vertices = []
+    indices = []
+    for axis in range(3):
+        other = [i for i in range(3) if i != axis]
+        for sign in (-1, 1):
+            start = len(vertices)
+            for i in range(steps + 1):
+                for j in range(steps + 1):
+                    p = np.zeros(3)
+                    p[axis] = sign * half[axis]
+                    p[other] = [(2 * i / steps - 1) * half[other[0]], (2 * j / steps - 1) * half[other[1]]]
+                    q = np.clip(p, -inner, inner)
+                    n = p - q
+                    n /= np.linalg.norm(n)
+                    vertices.append((*list(q + bevel * n), *list(n)))
+            for i in range(steps):
+                for j in range(steps):
+                    a = start + i * (steps + 1) + j
+                    b = a + steps + 1
+                    indices.extend((a, b, a + 1, a + 1, b, b + 1))
+    return finish(vertices, indices)

+def lathe(profile, sides=64, smooth=True):
+    """회전체의 실제 앞뒤·옆면을 만들며 다각 단면은 평면 법선을 보존합니다."""
+    vertices = []
+    indices = []
+    for row in range(len(profile) - 1):
+        radius, y = profile[row]
+        next_radius, next_y = profile[row + 1]
+        for j in range(sides):
+            a = math.tau * j / sides
+            b = math.tau * (j + 1) / sides
+            corners = [(radius * math.cos(a), y, radius * math.sin(a)), (radius * math.cos(b), y, radius * math.sin(b)), (next_radius * math.cos(b), next_y, next_radius * math.sin(b)), (next_radius * math.cos(a), next_y, next_radius * math.sin(a))]
+            start = len(vertices)
+            for k, p in enumerate(corners):
+                t = (a if k in (0, 3) else b) if smooth else (a + b) / 2
+                n = np.array([(next_y - y) * math.cos(t), radius - next_radius, (next_y - y) * math.sin(t)])
+                n /= np.linalg.norm(n)
+                vertices.append((*p, *n))
+            indices.extend((start, start + 1, start + 2, start, start + 2, start + 3))
+    return finish(vertices, indices)

+def tube(points, radius=3, closed=False, sides=14):
+    """3D 경로를 따라 원형 단면을 운반합니다. 코일은 실제 깊이를 가집니다."""
+    points = np.asarray(points, dtype=float)
+    vertices = []
+    indices = []
+    count = len(points)
+    for i, p in enumerate(points):
+        before = points[(i - 1) % count] if closed or i else points[0]
+        after = points[(i + 1) % count] if closed or i < count - 1 else points[-1]
+        tangent = after - before
+        tangent /= np.linalg.norm(tangent)
+        up = np.array([0.0, 1.0, 0.0]) if abs(tangent[1]) < 0.95 else np.array([1.0, 0.0, 0.0])
+        side = np.cross(tangent, up)
+        side /= np.linalg.norm(side)
+        vertical = np.cross(tangent, side)
+        for j in range(sides):
+            n = side * math.cos(math.tau * j / sides) + vertical * math.sin(math.tau * j / sides)
+            vertices.append((*list(p + radius * n), *list(n)))
+    for i in range(count if closed else count - 1):
+        for j in range(sides):
+            a = i * sides + j
+            b = (i + 1) % count * sides + j
+            c = (i + 1) % count * sides + (j + 1) % sides
+            d = i * sides + (j + 1) % sides
+            indices.extend((a, b, c, a, c, d))
+    return finish(vertices, indices)
