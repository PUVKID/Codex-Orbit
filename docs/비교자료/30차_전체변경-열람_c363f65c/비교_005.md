+class UsageAssetError(RuntimeError):
+    """조형 표시 오류를 호출자에게 전달하며 계정 조회 흐름과 분리합니다."""

+def asset_directory():

+def _identity(path):

+def _json_object(pairs):

+    for key, item in pairs:

+class UsageFrames:

+    def __init__(self, directory):

+        self.directory = Path(directory)
+        self.cache = OrderedDict()
+        self.archive = None
+        self._stream = None

+    def _prepare(self):

+        archive_path = self.directory / 'frames.zip'
+        if not 0 < manifest_path.stat().st_size <= MAX_MANIFEST_BYTES:
+            raise UsageAssetError('대시보드 조형 자산 매니페스트 크기가 허용 범위를 벗어났습니다.')
+        self.manifest = json.loads(manifest_path.read_text(encoding='utf-8'), object_pairs_hook=_json_object)

+                    'frame_pattern': 'frame_{:04d}.png', 'archive': 'frames.zip'}
+        if not isinstance(self.manifest, dict):

+        for key, value in expected.items():

+            if type(actual) is not type(value) or actual != value:

+        for key in ('geometry', 'lighting', 'source'):

+        digest = self.manifest.get('archive_sha256')

+        if not 0 < archive_path.stat().st_size <= MAX_ARCHIVE_BYTES:

+        self.identities = (_identity(manifest_path), _identity(archive_path))
+        self._stream = archive_path.open('rb')
+        hasher = hashlib.sha256()
+        while chunk := self._stream.read(1024 * 1024):

+        if hasher.hexdigest() != digest:

+    @staticmethod
+    def frame_index(elapsed):
+        if type(elapsed) not in (int, float):

+        try:
+            finite = math.isfinite(elapsed)
+        except OverflowError:
+            finite = False
+        if not finite:
+            raise UsageAssetError('대시보드 조형 프레임의 재생 시각이 유한한 숫자가 아닙니다.')

+
+    def frame(self, elapsed):
+        if self.archive is None:
+            raise UsageAssetError("조형 재생기가 닫혔습니다.")
+        try:
+            self._check_unchanged()
+        except OSError as error:
+            raise UsageAssetError("조형 자산 경로가 변경됐습니다.") from error
+        index = self.frame_index(elapsed)
+        if index not in self.cache:
+            try:
+                self.cache[index] = self._decode(f'frame_{index:04d}.png')
+            except Exception as error:
+                raise UsageAssetError('대시보드 조형 프레임 재생 실패: ' + str(error)) from error
+            while len(self.cache) > CACHE_LIMIT:
+                self.cache.popitem(last=False)
+        self.cache.move_to_end(index)
+        return self.cache[index]
+
+    @property
+    def cache_bytes(self):
+        return sum(image.sizeInBytes() for image in self.cache.values())
+
+    def close(self):
+        if self.archive is not None:
+            self.archive.close()
+            self.archive = None
+        if self._stream is not None:
+            self._stream.close()
+            self._stream = None
+        self.cache.clear()

+def rotate_point(point, angles):
+    """X·Y·Z 순서로 회전하며 깊이 좌표를 보존합니다."""
+    x, y, z = point
+    ax, ay, az = angles
+    y, z = y * math.cos(ax) - z * math.sin(ax), y * math.sin(ax) + z * math.cos(ax)
+    x, z = x * math.cos(ay) + z * math.sin(ay), -x * math.sin(ay) + z * math.cos(ay)
+    return (x * math.cos(az) - y * math.sin(az), x * math.sin(az) + y * math.cos(az), z)
+

+def project_point(point):
+    x, y, z = point
+    scale = 700 / (700 - z) * 74 / 77
+    return x * scale, -y * scale

+MOTION_PERIOD = 18
+RING_TURNS = (1, -2, 3)
+RING_AZIMUTHS = (8., 128., 248.)
+RING_AXIS_TILTS = (42., 46., 49.)
+RING_AXES = tuple((math.sin(math.radians(azimuth))*math.sin(math.radians(tilt)),
+                   -math.cos(math.radians(azimuth))*math.sin(math.radians(tilt)),
+                   math.cos(math.radians(tilt)))
+                  for azimuth, tilt in zip(RING_AZIMUTHS, RING_AXIS_TILTS))
+RING_RADII = (45., 51., 57.)
+BASE_ANGLES = tuple((math.radians(tilt), 0., math.radians(azimuth))
+                    for tilt, azimuth in zip((52., 58., 62.), RING_AZIMUTHS))

+def axis_rotate(point, axis, angle):
+    """로드리게스 식으로 고유 축 주위를 회전하며 공통 Z 변환을 쓰지 않습니다."""
+    length = math.sqrt(sum(v*v for v in axis))
+    u, v, w = (item/length for item in axis)
+    x, y, z = point
+    c, q = math.cos(angle), math.sin(angle)
+    dot = u*x + v*y + w*z
+    return (x*c + (v*z-w*y)*q + u*dot*(1-c),
+            y*c + (w*x-u*z)*q + v*dot*(1-c),
+            z*c + (u*y-v*x)*q + w*dot*(1-c))

+                                     BASE_ANGLES[index]), RING_AXES[index], angle) for n in range(count)]

+def orbit_segments(elapsed):
+    """제작 매니페스트와 일치하는 기하 검증용 메타입니다. GUI 드로잉에는 쓰지 않습니다."""

+def localized_region(data, language='ko'):
+    """검색 전 임시 표시에는 검증된 국내 사전 또는 실제 원문 지명을 사용합니다."""

+        if re.search('[가-힣]', value):

+        for suffix in (' special self-governing province', ' special self-governing city',

+def select_geocoded_region(location, data):
+    """동일 국가이면서 IP 추정 좌표에서30km 이내인 가장 가까운 검색 결과만 수락합니다."""
+    rows = data.get('results')
+    if not isinstance(rows, list) or len(rows) > 10:
+        return None
+    code = location.get('country_code', '')
+    if not isinstance(code, str) or not re.fullmatch('[A-Z]{2}', code):
+        return None
+    def valid(value, low, high):
+        return type(value) in (int, float) and low <= value <= high and math.isfinite(value)
+    latitude, longitude = location.get('latitude'), location.get('longitude')
+    if not valid(latitude, -90, 90) or not valid(longitude, -180, 180):
+        return None
+    closest = None
+    for row in rows:
+        if not isinstance(row, dict) or row.get('country_code') != code:
+            continue
+        lat, lon = row.get('latitude'), row.get('longitude')
+        name = row.get('name')
+        if not isinstance(name, str) or not name.strip() or not valid(lat, -90, 90) or not valid(lon, -180, 180):
+            continue
+        delta_lat, delta_lon = math.radians(lat-latitude), math.radians(lon-longitude)
+        arc = math.sin(delta_lat/2)**2 + math.cos(math.radians(lat))*math.cos(math.radians(latitude))*math.sin(delta_lon/2)**2
+        distance = 6371 * 2 * math.asin(math.sqrt(min(1, max(0, arc))))
+        if distance > 30 or (closest is not None and distance >= closest[0]):
+            continue
+        labels = []
+        for value in (name, row.get('admin1'), row.get('country')):
+            if isinstance(value, str):
+                value = ''.join(c for c in value if c.isprintable()).strip()[:80]
+                if value and value.casefold() not in [v.casefold() for v in labels]:
+                    labels.append(value)
+        closest = distance, ' · '.join(labels)
+    return closest[1] if closest else None
