+def filter_events(events, query='', kind='all', period='all', bookmarked=False,
+                  pinned=False, bookmarks=(), pins=(), oldest=False, now=None):
+    """원문 전후값을 비교하며 시간 미확인 기록은 날짜 필터에서 제외합니다."""
